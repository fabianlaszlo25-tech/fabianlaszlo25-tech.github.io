using UnityEngine;
using System.Runtime.InteropServices;

/// <summary>
/// Handles high-performance GPU instancing by bypassing traditional GameObject overhead.
/// Manages ComputeBuffers to push thousands of dynamic matrices directly to the graphics pipeline.
/// </summary>
public class GPUInstancedRenderer : MonoBehaviour
{
    [Header("Rendering Payload")]
    public Mesh instanceMesh;
    public Material instanceMaterial;
    
    private ComputeBuffer matrixBuffer;
    private ComputeBuffer argsBuffer;
    private Matrix4x4[] instanceTransforms;
    private uint[] args = new uint[5] { 0, 0, 0, 0, 0 };
    private int currentInstanceCount = 0;

    public void InitializePool(int capacity)
    {
        instanceTransforms = new Matrix4x4[capacity];
        
        matrixBuffer = new ComputeBuffer(capacity, Marshal.SizeOf(typeof(Matrix4x4)));
        argsBuffer = new ComputeBuffer(1, args.Length * sizeof(uint), ComputeBufferType.IndirectArguments);
        
        args[0] = (uint)instanceMesh.GetIndexCount(0);
        args[1] = (uint)capacity;
        args[2] = (uint)instanceMesh.GetIndexStart(0);
        args[3] = (uint)instanceMesh.GetBaseVertex(0);
        argsBuffer.SetData(args);
    }

    public void UpdateInstanceData(Matrix4x4[] updatedTransforms, int activeCount)
    {
        currentInstanceCount = activeCount;
        matrixBuffer.SetData(updatedTransforms, 0, 0, activeCount);
        instanceMaterial.SetBuffer("_InstanceMatrices", matrixBuffer);
    }

    private void Update()
    {
        if (currentInstanceCount > 0 && matrixBuffer != null)
        {
            // Dispatches a single draw call for all active instances, drastically reducing CPU overhead.
            Graphics.DrawMeshInstancedIndirect(
                instanceMesh, 
                0, 
                instanceMaterial, 
                new Bounds(Vector3.zero, new Vector3(1000f, 1000f, 1000f)), 
                argsBuffer
            );
        }
    }

    private void OnDestroy()
    {
        matrixBuffer?.Release();
        argsBuffer?.Release();
    }
}
              

using System;
using System.Collections.Concurrent;
using System.Collections.Generic;

/// <summary>
/// A lock-free, zero-allocation event aggregator designed for decoupled MVVM state propagation.
/// Ensures that domain models can trigger view updates without creating hard references.
/// </summary>
public static class ThreadSafeEventBus
{
    private static readonly ConcurrentDictionary<Type, List<Delegate>> _subscribers = new();

    public static void Subscribe<TEvent>(Action<TEvent> callback) where TEvent : struct
    {
        Type eventType = typeof(TEvent);
        
        _subscribers.AddOrUpdate(
            eventType,
            new List<Delegate> { callback },
            (key, existingList) =>
            {
                lock (existingList)
                {
                    existingList.Add(callback);
                }
                return existingList;
            });
    }

    public static void Unsubscribe<TEvent>(Action<TEvent> callback) where TEvent : struct
    {
        if (_subscribers.TryGetValue(typeof(TEvent), out var existingList))
        {
            lock (existingList)
            {
                existingList.Remove(callback);
            }
        }
    }

    public static void Publish<TEvent>(in TEvent eventPayload) where TEvent : struct
    {
        if (_subscribers.TryGetValue(typeof(TEvent), out var existingList))
        {
            // Create a local copy of the invocation list to ensure thread safety during enumeration
            Delegate[] callbacks;
            lock (existingList)
            {
                callbacks = existingList.ToArray();
            }

            foreach (Delegate callback in callbacks)
            {
                ((Action<TEvent>)callback).Invoke(eventPayload);
            }
        }
    }
}
