public class SaveManager : MonoBehaviour
{
    public static SaveManager Instance;

    [Header("Settings")]
    public string mainMenuSceneName = "00Pre-Stuff";

    private float autosaveTimer = 0f;
    private float autosaveInterval = 300f;

    private GameData currentData = new GameData();
    public List activeSaveables = new List();

    void Awake()
    {
        if (Instance == null)
        {
            Instance = this;
            DontDestroyOnLoad(gameObject);
        }
        else
        {
            Destroy(gameObject);
        }
    }

    void Start()
    {
        LoadGame();
    }

    void Update()
    {
        autosaveTimer += Time.deltaTime;
        if (autosaveTimer >= autosaveInterval)
        {
            SaveGame();
            autosaveTimer = 0f;
        }
    }

    void OnApplicationQuit()
    {
        SaveGame();
    }

    private string GetSavePath()
    {
        string saveFolder = Path.Combine(Directory.GetParent(Application.dataPath).FullName, "Saves");
        if (!Directory.Exists(saveFolder)) Directory.CreateDirectory(saveFolder);
        return Path.Combine(saveFolder, "SaveData.json");
    }

    public void Register(Saveable s)
    {
        if (currentData.destroyedObjectIds.Contains(s.uniqueId))
        {
            Destroy(s.gameObject);
            return;
        }

        if (!activeSaveables.Contains(s))
        {
            activeSaveables.Add(s);

            ObjectData savedData = currentData.objectDatas.Find(x => x.id == s.uniqueId);
            if (savedData != null) s.ApplyObjectData(savedData);
        }
    }

    public void Deregister(Saveable s) { activeSaveables.Remove(s); }

    public void SaveGame()
    {
        if (SceneManager.GetActiveScene().name == mainMenuSceneName)
        {
            return;
        }

        currentData.objectDatas.Clear();
        currentData.lastSavedScene = SceneManager.GetActiveScene().name;

        foreach (Saveable s in activeSaveables)
        {
            currentData.objectDatas.Add(s.GetObjectData());
        }

        string json = JsonUtility.ToJson(currentData, true);
        File.WriteAllText(GetSavePath(), json);
        Debug.Log("Game Saved to: " + GetSavePath());
    }
    public void RecordDestroyed(string id)
    {
        if (!currentData.destroyedObjectIds.Contains(id))
        {
            currentData.destroyedObjectIds.Add(id);
        }
    }
    public void LoadGame()
    {
        string path = GetSavePath();
        if (File.Exists(path))
        {
            string json = File.ReadAllText(path);
            currentData = JsonUtility.FromJson(json);

            foreach (Saveable s in activeSaveables)
            {
                ObjectData savedData = currentData.objectDatas.Find(x => x.id == s.uniqueId);
                if (savedData != null) s.ApplyObjectData(savedData);
            }
        }
        else
        {
            currentData = new GameData();
        }
    }

    public void WipeSaveForNewGame()
    {
        currentData = new GameData(); 

        string path = GetSavePath();
        if (File.Exists(path))
        {
            File.Delete(path); 
        }
        Debug.Log("Save file wiped for a new game.");
    }
}

using UnityEngine;

public class Saveable : MonoBehaviour
{
    [Tooltip("Right-click the component header and select 'Generate ID'")]
    public string uniqueId;

    [ContextMenu("Generate ID")]
    private void GenerateId()
    {
        uniqueId = System.Guid.NewGuid().ToString();
    }

    void Start()
    {
        if (string.IsNullOrEmpty(uniqueId)) GenerateId();

        if (SaveManager.Instance != null) SaveManager.Instance.Register(this);
    }

    void OnDestroy()
    {
        if (SaveManager.Instance != null)
        {
            SaveManager.Instance.Deregister(this);

            if (gameObject.scene.isLoaded)
            {
                SaveManager.Instance.RecordDestroyed(uniqueId);
            }
        }
    }

    public ObjectData GetObjectData()
    {
        ObjectData data = new ObjectData
        {
            id = uniqueId,
            position = transform.position,
            rotation = transform.rotation,
            scale = transform.localScale,
            isActive = gameObject.activeSelf
        };

        ElevatorSystem elevator = GetComponent();
        if (elevator != null)
        {
            data.customData = elevator.GetSaveState();
        }

        DialogueManager dialogueManager = GetComponent();
        if (dialogueManager != null)
        {
            data.customData = dialogueManager.GetSaveState();
        }

        return data;
    }

    public void ApplyObjectData(ObjectData data)
    {
        transform.position = data.position;
        transform.rotation = data.rotation;
        transform.localScale = data.scale;

        gameObject.SetActive(data.isActive);

        if (!string.IsNullOrEmpty(data.customData))
        {
            ElevatorSystem elevator = GetComponent();
            if (elevator != null)
            {
                elevator.LoadSaveState(data.customData);
            }

            DialogueManager dialogueManager = GetComponent();
            if (dialogueManager != null)
            {
                dialogueManager.LoadSaveState(data.customData);
            }
        }
    }
}

public class SettingsManager : MonoBehaviour
{
    [Header("UI References")]
    public TMP_Dropdown resolutionDropdown;
    public TMP_Dropdown displayModeDropdown;
    public Slider volumeSlider;

    [Tooltip("Set this slider to use Whole Numbers from 0 to 3 in the Inspector.")]
    public Slider qualitySlider;

    [Header("New Settings")]
    public Slider sensitivitySlider;

    private List filteredResolutions;

    void Start()
    {
        Resolution[] allResolutions = Screen.resolutions;
        filteredResolutions = new List();
        resolutionDropdown.ClearOptions();
        List options = new List();
        int currentResIndex = 0;

        for (int i = 0; i < allResolutions.Length; i++)
        {
            if (filteredResolutions.Count == 0 ||
                filteredResolutions[filteredResolutions.Count - 1].width != allResolutions[i].width ||
                filteredResolutions[filteredResolutions.Count - 1].height != allResolutions[i].height)
            {
                filteredResolutions.Add(allResolutions[i]);
                string option = allResolutions[i].width + " x " + allResolutions[i].height;
                options.Add(option);

                if (allResolutions[i].width == PlayerPrefs.GetInt("ResWidth", Screen.currentResolution.width) &&
                    allResolutions[i].height == PlayerPrefs.GetInt("ResHeight", Screen.currentResolution.height))
                {
                    currentResIndex = filteredResolutions.Count - 1;
                }
            }
        }

        resolutionDropdown.AddOptions(options);
        resolutionDropdown.value = currentResIndex;
        resolutionDropdown.RefreshShownValue();

        
        displayModeDropdown.value = PlayerPrefs.GetInt("DisplayMode", 1);
        volumeSlider.value = PlayerPrefs.GetFloat("MasterVolume", 1f);

        if (qualitySlider != null)
            qualitySlider.value = PlayerPrefs.GetInt("QualityLevel", 3); 

        if (sensitivitySlider != null)
            sensitivitySlider.value = PlayerPrefs.GetFloat("MouseSensitivity", 5.5f);

        ApplyQualitySettings();
        ApplyAudioSettings();

        resolutionDropdown.onValueChanged.AddListener(delegate { SaveResolution(); });
        displayModeDropdown.onValueChanged.AddListener(delegate { SaveDisplayMode(); });
        volumeSlider.onValueChanged.AddListener(delegate { ApplyAudioSettings(); });

        if (qualitySlider != null)
            qualitySlider.onValueChanged.AddListener(delegate { ApplyQualitySettings(); });

        if (sensitivitySlider != null)
            sensitivitySlider.onValueChanged.AddListener(delegate { SaveSensitivity(); });
    }

    public void SaveResolution()
    {
        Resolution res = filteredResolutions[resolutionDropdown.value];
        PlayerPrefs.SetInt("ResWidth", res.width);
        PlayerPrefs.SetInt("ResHeight", res.height);
        PlayerPrefs.Save();
    }

    public void SaveDisplayMode()
    {
        PlayerPrefs.SetInt("DisplayMode", displayModeDropdown.value);
        PlayerPrefs.Save();
    }

    public void SaveSensitivity()
    {
        PlayerPrefs.SetFloat("MouseSensitivity", sensitivitySlider.value);
        PlayerPrefs.Save();
    }

    public void ApplyAudioSettings()
    {
        float vol = volumeSlider.value;
        PlayerPrefs.SetFloat("MasterVolume", vol);
        PlayerPrefs.Save();
        AudioListener.volume = vol;
    }

    public void ApplyQualitySettings()
    {
        UniversalRenderPipelineAsset urpAsset = (UniversalRenderPipelineAsset)GraphicsSettings.currentRenderPipeline;
        if (urpAsset == null) return;

        int qualityLevel = (int)qualitySlider.value;
        PlayerPrefs.SetInt("QualityLevel", qualityLevel);

        
        switch (qualityLevel)
        {
            case 0: // Low
                urpAsset.renderScale = 0.5f;
                urpAsset.shadowDistance = 0f; 
                urpAsset.msaaSampleCount = 1; 
                break;

            case 1: // Medium
                urpAsset.renderScale = 0.75f;
                urpAsset.shadowDistance = 50f;
                urpAsset.mainLightShadowmapResolution = 1024; 
                urpAsset.msaaSampleCount = 2; 
                break;

            case 2: // High
                urpAsset.renderScale = 1.0f;
                urpAsset.shadowDistance = 50f;
                urpAsset.mainLightShadowmapResolution = 4096; 
                urpAsset.msaaSampleCount = 8; 
                break;

            case 3: // Max
                urpAsset.renderScale = 1.2f;
                urpAsset.shadowDistance = 100f;
                urpAsset.mainLightShadowmapResolution = 4096; 
                urpAsset.msaaSampleCount = 16; 
                break;
        }

        PlayerPrefs.Save();
    }

public class LauncherManager : MonoBehaviour
{
    [Header("Scene Settings")]
    public string gameSceneName = "MainGameScene";

    [Header("UI References")]
    [Tooltip("Drag the UI Panel here that asks 'Are you sure?' when overwriting a save.")]
    public GameObject overwriteWarningPanel;

    [Tooltip("Drag your Continue button here so it can be disabled if no save exists.")]
    public GameObject continueButton;

    void Awake()
    {
        SetScreenResolution();
    }

    void Start()
    {
        if (overwriteWarningPanel != null) overwriteWarningPanel.SetActive(false);

        if (continueButton != null)
        {
            continueButton.SetActive(File.Exists(GetSavePath()));
        }
    }

    private string GetSavePath()
    {
        string saveFolder = Path.Combine(Directory.GetParent(Application.dataPath).FullName, "Saves");
        if (!Directory.Exists(saveFolder)) Directory.CreateDirectory(saveFolder);

        return Path.Combine(saveFolder, "SaveData.json");
    }


    public void OnNewGameClicked()
    {
        if (File.Exists(GetSavePath()))
        {
            if (overwriteWarningPanel != null) overwriteWarningPanel.SetActive(true);
        }
        else
        {
            PlayerPrefs.SetInt("HasPlayedIntro", 0);
            PlayerPrefs.Save();

            StartNewGame();
        }
    }

    public void ConfirmNewGame()
    {
        string path = GetSavePath();
        if (File.Exists(path))
        {
            File.Delete(path);
        }

        if (SaveManager.Instance != null)
        {
            SaveManager.Instance.WipeSaveForNewGame();
        }

        PlayerPrefs.SetInt("HasPlayedIntro", 0);
        PlayerPrefs.Save();

        StartNewGame();
    }

    public void CancelNewGame()
    {
        if (overwriteWarningPanel != null) overwriteWarningPanel.SetActive(false);
    }

    private void StartNewGame()
    {
        SetScreenResolution();
        Debug.Log("Starting fresh game in: " + gameSceneName);
        SceneManager.LoadScene(gameSceneName);
    }


    public void ContinueGame()
    {
        SetScreenResolution();

        string pathToLoad = GetSavePath();
        string sceneToLoad = gameSceneName;

        if (File.Exists(pathToLoad))
        {
            string json = File.ReadAllText(pathToLoad);
            GameData savedData = JsonUtility.FromJson(json);

            if (!string.IsNullOrEmpty(savedData.lastSavedScene))
            {
                sceneToLoad = savedData.lastSavedScene;
                Debug.Log("Found save file! Loading saved scene: " + sceneToLoad);
            }
        }
        else
        {
            Debug.LogWarning("Tried to continue, but no save found! Starting new game instead.");
        }

        SceneManager.LoadScene(sceneToLoad);
    }


    private void SetScreenResolution()
    {
        Resolution[] allResolutions = Screen.resolutions;
        Resolution nativeRes = allResolutions[allResolutions.Length - 1];

        FullScreenMode forcedMode = FullScreenMode.FullScreenWindow;

        Screen.SetResolution(nativeRes.width, nativeRes.height, forcedMode);

        PlayerPrefs.SetInt("ResWidth", nativeRes.width);
        PlayerPrefs.SetInt("ResHeight", nativeRes.height);
        PlayerPrefs.SetInt("DisplayMode", 1); 
        PlayerPrefs.Save();
    }

    public void QuitLauncher() => Application.Quit();
}

public class PauseManager : MonoBehaviour
{
    [Header("Pause Camera Setup")]
    public Camera pauseCamera;

    [Header("UI Panels")]
    public GameObject pauseMenuPanel;
    public GameObject settingsPanel;
    public GameObject interactCanvas;

    [Header("Interaction Managers")]
    public DualSceneManager dualSceneManager;
    public PlayerInspector playerInspector;

    [Header("World Space Canvases")]
    public Canvas[] worldSpaceCanvases;

    private Camera activePlayerCamera;

    private float toggleCooldown = 0.2f;
    private float lastToggleTime = 0f;

    void Start()
    {
        pauseMenuPanel.SetActive(false);
        if (settingsPanel != null) settingsPanel.SetActive(false);

        if (pauseCamera != null)
        {
            pauseCamera.gameObject.SetActive(true);
            pauseCamera.enabled = false;
        }

        Time.timeScale = 1f;
        Cursor.lockState = CursorLockMode.Locked;
        Cursor.visible = false;

        if (interactCanvas != null) interactCanvas.SetActive(true);
    }

    public void LoadMainMenu()
    {
        Time.timeScale = 1f;

        if (SaveManager.Instance != null)
        {
            SaveManager.Instance.SaveGame();
        }

        SceneManager.LoadScene("00Pre-Stuff");
    }

    void Update()
    {
        if (Input.GetKeyDown(KeyCode.Escape) && Time.unscaledTime > lastToggleTime + toggleCooldown)
        {
            lastToggleTime = Time.unscaledTime;

            if ((dualSceneManager != null && (dualSceneManager.isAtComputer || dualSceneManager.isTransitioning)) ||
                (playerInspector != null && (playerInspector.isInspecting || playerInspector.isTransitioning)))
            {
                return;
            }

            if (pauseMenuPanel.activeSelf || (settingsPanel != null && settingsPanel.activeSelf))
            {
                ResumeGame();
            }
            else
            {
                PauseGame();
            }
        }
    }

    void LateUpdate()
    {
        if (pauseMenuPanel.activeSelf || (settingsPanel != null && settingsPanel.activeSelf))
        {
            Cursor.lockState = CursorLockMode.None;
            Cursor.visible = true;
        }
    }

    public void PauseGame()
    {
        Time.timeScale = 0f;
        pauseMenuPanel.SetActive(true);

        if (interactCanvas != null) interactCanvas.SetActive(false);

        if (Camera.main != null && Camera.main != pauseCamera)
        {
            activePlayerCamera = Camera.main;
            activePlayerCamera.enabled = false;
        }

        if (pauseCamera != null)
        {
            pauseCamera.gameObject.SetActive(true);
            pauseCamera.enabled = true;
        }

        foreach (Canvas canvas in worldSpaceCanvases)
        {
            if (canvas != null) canvas.worldCamera = pauseCamera;
        }

        TogglePlayerScripts(false);
    }

    public void ResumeGame()
    {
        Time.timeScale = 1f;
        pauseMenuPanel.SetActive(false);
        if (settingsPanel != null) settingsPanel.SetActive(false);

        if (pauseCamera != null) pauseCamera.enabled = false;
        if (activePlayerCamera != null) activePlayerCamera.enabled = true;

        TogglePlayerScripts(true);

        Cursor.lockState = CursorLockMode.Locked;
        Cursor.visible = false;
    }

    private void TogglePlayerScripts(bool turnOn)
    {
        PlayerInspector inspector = FindObjectOfType();
        ElevatorButtonScanner scanner = FindObjectOfType();

        bool enableOuter = turnOn;
        bool enableInner = false;

        if (inspector != null)
        {
            if (turnOn && inspector.isInspecting)
            {
                enableOuter = false;
                enableInner = true;
            }
        }

        MSMO_PlayerController playerCtrl = FindObjectOfType();
        if (playerCtrl != null) playerCtrl.enabled = enableOuter;

        MSMoveObjects moveObj = FindObjectOfType();
        if (moveObj != null) moveObj.enabled = enableOuter;

        if (inspector != null) inspector.enabled = turnOn;
        if (scanner != null) scanner.enabled = turnOn;

        if (dualSceneManager != null) dualSceneManager.enabled = turnOn;

        InnerPlayerControl innerCtrl = FindObjectOfType();
        if (innerCtrl != null) innerCtrl.SetControlsActive(enableInner);
    }

    public void OpenSettings()
    {
        pauseMenuPanel.SetActive(false);
        if (settingsPanel != null) settingsPanel.SetActive(true);
    }

    public void CloseSettings()
    {
        if (settingsPanel != null) settingsPanel.SetActive(false);
        pauseMenuPanel.SetActive(true);
    }

    public void QuitGame()
    {
        Time.timeScale = 1f;

        if (SaveManager.Instance != null)
        {
            SaveManager.Instance.SaveGame();
        }

        Application.Quit();
    }
}

using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;

public class DatamoshFeature : ScriptableRendererFeature
{
    [System.Serializable]
    public class Settings
    {
        public Shader unifiedShader;
        public LayerMask appearLayerMask;
    }

    public Settings settings = new Settings();
    private Material material;
    private MaskPass maskPass;
    private MoshPass moshPass;
    private RTHandle maskBuffer;
    private RTHandle tempBuffer;

    public static bool requestClear = false;

    public override void Create()
    {
        if (settings.unifiedShader == null) return;
        material = CoreUtils.CreateEngineMaterial(settings.unifiedShader);

        maskPass = new MaskPass(material, settings.appearLayerMask);
        maskPass.renderPassEvent = RenderPassEvent.BeforeRenderingPostProcessing;

        moshPass = new MoshPass(material);
        moshPass.renderPassEvent = RenderPassEvent.BeforeRenderingPostProcessing;
    }

    public override void AddRenderPasses(ScriptableRenderer renderer, ref RenderingData renderingData)
    {
        if (material == null) return;
        if (renderingData.cameraData.cameraType != CameraType.Game) return;
        if (renderingData.cameraData.camera.targetTexture == null) return;

        var desc = renderingData.cameraData.cameraTargetDescriptor;
        if (desc.width <= 0 || desc.height <= 0) return;

        desc.depthBufferBits = 0;
        desc.graphicsFormat = UnityEngine.Experimental.Rendering.GraphicsFormat.R8G8B8A8_UNorm;

        RenderingUtils.ReAllocateIfNeeded(ref maskBuffer, desc, FilterMode.Point, TextureWrapMode.Clamp, name: "_DatamoshMask");
        RenderingUtils.ReAllocateIfNeeded(ref tempBuffer, desc, FilterMode.Bilinear, TextureWrapMode.Clamp, name: "_DatamoshTemp");

        maskPass.Setup(maskBuffer);
        moshPass.Setup(maskBuffer, tempBuffer);

        renderer.EnqueuePass(maskPass);
        renderer.EnqueuePass(moshPass);
    }

    protected override void Dispose(bool disposing)
    {
        CoreUtils.Destroy(material);
        maskBuffer?.Release();
        tempBuffer?.Release();
        if (moshPass != null) moshPass.Cleanup();
    }

    class MaskPass : ScriptableRenderPass
    {
        private Material material;
        private LayerMask appearLayer;
        private RTHandle maskBuffer;

        public MaskPass(Material mat, LayerMask layer) { material = mat; appearLayer = layer; }
        public void Setup(RTHandle mask) { maskBuffer = mask; }

        public override void Execute(ScriptableRenderContext context, ref RenderingData renderingData)
        {
            CommandBuffer cmd = CommandBufferPool.Get("Datamosh Mask");
            CoreUtils.SetRenderTarget(cmd, maskBuffer, ClearFlag.Color, Color.black);

            DrawingSettings drawSettings = CreateDrawingSettings(new ShaderTagId("UniversalForward"), ref renderingData, SortingCriteria.CommonOpaque);
            drawSettings.SetShaderPassName(1, new ShaderTagId("UniversalForwardOnly"));
            drawSettings.SetShaderPassName(2, new ShaderTagId("LightweightForward"));
            drawSettings.SetShaderPassName(3, new ShaderTagId("SRPDefaultUnlit"));
            drawSettings.SetShaderPassName(4, new ShaderTagId("DepthOnly"));

            drawSettings.overrideMaterial = material;
            drawSettings.overrideMaterialPassIndex = 1;

            FilteringSettings filterSettings = new FilteringSettings(RenderQueueRange.all, appearLayer);

            context.ExecuteCommandBuffer(cmd);
            cmd.Clear();
            context.DrawRenderers(renderingData.cullResults, ref drawSettings, ref filterSettings);

            context.ExecuteCommandBuffer(cmd);
            CommandBufferPool.Release(cmd);
        }
    }

    class MoshPass : ScriptableRenderPass
    {
        private Material material;
        private RTHandle maskBuffer, tempBuffer;
        private RenderTexture historyRT;
        private RTHandle historyHandle;

        public MoshPass(Material mat) { material = mat; }
        public void Setup(RTHandle mask, RTHandle temp) { maskBuffer = mask; tempBuffer = temp; }

        public override void OnCameraSetup(CommandBuffer cmd, ref RenderingData renderingData)
        {
            ConfigureInput(ScriptableRenderPassInput.Color | ScriptableRenderPassInput.Depth | ScriptableRenderPassInput.Motion);
        }

        public override void Execute(ScriptableRenderContext context, ref RenderingData renderingData)
        {
            CommandBuffer cmd = CommandBufferPool.Get("Datamosh Blit");

            var cameraTarget = renderingData.cameraData.renderer.cameraColorTargetHandle;

            if (cameraTarget == null || cameraTarget.rt == null) { CommandBufferPool.Release(cmd); return; }
            if (tempBuffer == null || tempBuffer.rt == null) { CommandBufferPool.Release(cmd); return; }

            var desc = renderingData.cameraData.cameraTargetDescriptor;
            if (historyRT == null || historyRT.width != desc.width || historyRT.height != desc.height)
            {
                if (historyRT != null) historyRT.Release();
                historyHandle?.Release();

                historyRT = new RenderTexture(desc.width, desc.height, 0, UnityEngine.Experimental.Rendering.GraphicsFormat.R8G8B8A8_UNorm);
                historyRT.name = "Datamosh_WorkTex_Internal";
                historyRT.Create();
                historyHandle = RTHandles.Alloc(historyRT);
            }

            if (historyHandle == null || historyHandle.rt == null) { CommandBufferPool.Release(cmd); return; }

            if (DatamoshFeature.requestClear)
            {
                CoreUtils.SetRenderTarget(cmd, historyHandle, ClearFlag.Color, Color.clear);
                DatamoshFeature.requestClear = false; 
            }

            cmd.SetGlobalTexture("_WorkTex", historyHandle);
            cmd.SetGlobalTexture("_MaskTex", maskBuffer);

            Blitter.BlitCameraTexture(cmd, cameraTarget, tempBuffer, material, 0);
            Blitter.BlitCameraTexture(cmd, tempBuffer, historyHandle, material, 2);
            Blitter.BlitCameraTexture(cmd, tempBuffer, cameraTarget, material, 2);

            context.ExecuteCommandBuffer(cmd);
            CommandBufferPool.Release(cmd);
        }

        public void Cleanup()
        {
            if (historyRT != null) historyRT.Release();
            historyHandle?.Release();
        }
    }

Shader "Hidden/DatamoshUnified"
{
    SubShader
    {
        Tags { "RenderType" = "Opaque" "RenderPipeline" = "UniversalPipeline" }
        Cull Off ZWrite Off ZTest Always

        Pass
        {
            Name "Datamosh Blit"
            ColorMask RGBA
            Blend Off
            
            HLSLPROGRAM
            #pragma vertex Vert
            #pragma fragment Frag
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"
            #include "Packages/com.unity.render-pipelines.core/Runtime/Utilities/Blit.hlsl"

            TEXTURE2D_X(_WorkTex);            SAMPLER(sampler_WorkTex);
            TEXTURE2D_X(_MotionVectorTexture);SAMPLER(sampler_MotionVectorTexture);
            TEXTURE2D_X(_MaskTex);            SAMPLER(sampler_MaskTex);

            float _BlockSize;
            float _Scale;
            float _Blend;
            float _SpatialErosion;
            float _IsIFrame;
            float _IsDisappearing; 
            float _MaxSmearPixels;
            float _AppearErosion;

            half4 Frag(Varyings input) : SV_Target
            {
                float2 uv = input.texcoord;
                float2 texel = 1.0 / _ScreenParams.xy;
                
                half4 realColor = SAMPLE_TEXTURE2D_X(_BlitTexture, sampler_LinearClamp, uv);
                float currentMask = SAMPLE_TEXTURE2D_X(_MaskTex, sampler_LinearClamp, uv).r;

                if (_IsIFrame > 0.5) 
                {
                    return half4(realColor.rgb, currentMask);
                }
                
                float2 blockUv = floor(uv * _ScreenParams.xy / _BlockSize) * _BlockSize * texel;
                
                float2 smoothMv = SAMPLE_TEXTURE2D_X(_MotionVectorTexture, sampler_MotionVectorTexture, uv).rg;
                float2 blockMv = SAMPLE_TEXTURE2D_X(_MotionVectorTexture, sampler_MotionVectorTexture, blockUv).rg;
                float2 mv = lerp(smoothMv, blockMv, 0.5) * _Scale; 

                float2 pixelOffset = mv * _ScreenParams.xy;          
                float pixelDist = length(pixelOffset);
                float maxPixel = _MaxSmearPixels;

                if (pixelDist > maxPixel)
                {
                      return half4(realColor.rgb, 0.0);   
                }

/// <summary>
/// A lock-free, zero-allocation event aggregator designed for decoupled MVVM state propagation.
/// Ensures that domain models can trigger view updates without creating hard references.
/// </summary>
public static class ThreadSafeEventBus
{
    private static readonly ConcurrentDictionary<Type, List<Delegate>> _subscribers = new();

    public static void Subscribe<TEvent>(Action<TEvent> callback) where TEvent : struct
    {
        Type eventType = typeof(TEvent);
        
        _subscribers.AddOrUpdate(
            eventType,
            new List<Delegate> { callback },
            (key, existingList) =>
            {
                lock (existingList)
                {
                    existingList.Add(callback);
                }
                return existingList;
            });
    }

    public static void Unsubscribe<TEvent>(Action<TEvent> callback) where TEvent : struct
    {
        if (_subscribers.TryGetValue(typeof(TEvent), out var existingList))
        {
            lock (existingList)
            {
                existingList.Remove(callback);
            }
        }
    }

    public static void Publish<TEvent>(in TEvent eventPayload) where TEvent : struct
    {
        if (_subscribers.TryGetValue(typeof(TEvent), out var existingList))
        {
            // Create a local copy of the invocation list to ensure thread safety during enumeration
            Delegate[] callbacks;
            lock (existingList)
            {
                callbacks = existingList.ToArray();
            }

            foreach (Delegate callback in callbacks)
            {
                ((Action<TEvent>)callback).Invoke(eventPayload);
            }
        }
    }
}

            else if (Input.GetKey(KeyCode.DownArrow))

            currentMonitorZoom = Mathf.Lerp(currentMonitorZoom, targetMonitorZoom, Time.deltaTime * zoomSmoothness);

        Quaternion targetRot = movingToMonitor ? monitorCameraSpot.rotation : originalCamParent.rotation * originalCamLocalRot;

        float t = 0f;
        while (t < 1f)
        {
            t += Time.deltaTime * transitionSpeed;
            float smoothStep = Mathf.SmoothStep(0, 1, t);

            mainCamera.position = Vector3.Lerp(mainCamera.position, targetPos, smoothStep);
            mainCamera.rotation = Quaternion.Slerp(mainCamera.rotation, targetRot, smoothStep);

            yield return null;
        }

        mainCamera.position = targetPos;
        mainCamera.rotation = targetRot;

        if (movingToMonitor)
        {
            if (innerPlayerControl != null) innerPlayerControl.SetControlsActive(true);
        }
        else
        {
            mainCamera.SetParent(originalCamParent);
            if (outerPlayerMovement != null) outerPlayerMovement.enabled = true;

            if (irlMoveObjects != null) irlMoveObjects.forceHideUI = false;
            if (pauseManagerObject != null) pauseManagerObject.SetActive(true);
        }

        isTransitioning = false;

        StartCoroutine(SwitchSceneRoutine(nextIndex));

    public void SwitchToPreviousScene()

    }

    {

    void LateUpdate()
    {
        if (!isAnimatingAction && anim != null && anim.runtimeAnimatorController != null)
        {
            string targetState = (currentState == FigureState.Placed || currentState == FigureState.InBattle) ? stateBattleIdle : stateIdle;

            if (!anim.GetCurrentAnimatorStateInfo(0).IsName(targetState) || anim.IsInTransition(0))
            {

            }
        }

    {
        currentState = isHovered ? FigureState.Hovered : FigureState.Idle;
    }

    {
        currentState = newState;

        Collider[] allColliders = GetComponentsInChildren();
        foreach (Collider col in allColliders)
        {
            col.enabled = (currentState != FigureState.Selected);
        }
    }

    public IEnumerator ExecuteAttack(Figure target, System.Action onGlobalDamageDealt)
    {
        if (currentState == FigureState.Dead) yield break;

        isAnimatingAction = true;
        ToggleEffects(attackEffects, true);
        PlaySafeAnim(stateAttack);

        yield return new WaitForSeconds(attackHitFrame);

        int overkillDamage = 0;

        if (target != null && target.currentState != FigureState.Dead)
        {
            overkillDamage = target.TakeDamage(attackDamage);

            if (target.hasThorns && target.currentState != FigureState.Dead)
            {
                this.TakeDamage(target.thornsDamage);
            }
        }
        else
        {
            overkillDamage = attackDamage;
        }

        onGlobalDamageDealt?.Invoke(overkillDamage);

        float remainingTime = Mathf.Max(0, attackTotalDuration - attackHitFrame);
        yield return new WaitForSeconds(remainingTime);

        ToggleEffects(attackEffects, false);

        isAnimatingAction = false;
        if (currentState != FigureState.Dead)

            PlaySafeAnim(stateBattleIdle);
        }
    }

        int overkill = Mathf.Max(0, amount - currentHP);

        currentHP -= amount;
        UpdateStatUI();

        if (currentHP <= 0)
        {
            Die();
        }
        else
        {
            StartCoroutine(HitVisualsRoutine());
        }

        return overkill;
    }

        }

    public class GameLogic
    {
        private readonly Random rnd = new();
        private readonly IGameDataAccess _dataAccess;
        public int N { get; private set; } = 11;
        public Point Player { get; private set; }
        public List Enemies { get; private set; } = new();
        public HashSet Mines { get; private set; } = new();
        public int ElapsedSeconds { get; private set; }
        public bool Paused { get; private set; }
        public bool GameOver { get; private set; }

        public event EventHandler? OnStateChanged;
        public event EventHandler? OnTimeChanged;
        public event EventHandler? OnGameOver;

        private IGameTimer enemyTimer;
        private IGameTimer clockTimer;

        public GameLogic(IGameDataAccess dataAccess)
        {
            _dataAccess = dataAccess ?? throw new ArgumentNullException(nameof(dataAccess));
            enemyTimer = new GameTimerAggregation();
            clockTimer = new GameTimerAggregation();

            enemyTimer.Interval = 400;
            enemyTimer.Elapsed += (s, e) => MoveEnemies();
            clockTimer.Interval = 1000;
            clockTimer.Elapsed += (s, e) =>
            {
                if (!Paused && !GameOver)
                {
                    ElapsedSeconds++;
                    OnTimeChanged?.Invoke(this, EventArgs.Empty);
                }
            };
        }

        public void Start()
        {
            NewGame(N);
        }
        

        
        private void LoadFromPersistenceData(GameData data)
        {
            N = data.N;
            Player = new Point(data.Player.X, data.Player.Y);
            Enemies = data.Enemies.Select(p => new Point(p.X, p.Y)).ToList();
            Mines = new HashSet(data.Mines.Select(p => new Point(p.X, p.Y)));
            ElapsedSeconds = data.ElapsedSeconds;
            Paused = true;
            GameOver = false;
            OnStateChanged?.Invoke(this, EventArgs.Empty);
        }

        
        private GameData ConvertToPersistenceData()
        {
            return new GameData
            {
                N = N,
                Player = new Persistence.Position(Player.X, Player.Y),
                Enemies = Enemies.Select(p => new Persistence.Position(p.X, p.Y)).ToList(),
                Mines = Mines.Select(p => new Persistence.Position(p.X, p.Y)).ToList(),
                ElapsedSeconds = ElapsedSeconds
            };
        }

        public void NewGame(int size)
        {
            N = size;
            ElapsedSeconds = 0;
            GameOver = false;
            Paused = false;
            Mines.Clear();
            Enemies.Clear();

            Player = new Point(N / 2, 0);
            Enemies.Add(new Point(0, N - 1));
            Enemies.Add(new Point(N - 1, N - 1));

            int mineCount = Math.Max(1, N / 3);
            while (Mines.Count < mineCount)
            {
                var p = new Point(rnd.Next(0, N), rnd.Next(0, N));
                if (p == Player || Enemies.Contains(p)) continue;
                Mines.Add(p);
            }

            enemyTimer.Start();
            clockTimer.Start();
            OnStateChanged?.Invoke(this, EventArgs.Empty);
        }

        public void MovePlayer(Point delta)
        {
            if (Paused || GameOver) return;
            var newPos = new Point(Player.X + delta.X, Player.Y + delta.Y);
            if (newPos.X < 0 || newPos.X >= N || newPos.Y < 0 || newPos.Y >= N) return;
            Player = newPos;
            CheckCollisions();
            OnStateChanged?.Invoke(this, EventArgs.Empty);
        }

        private void MoveEnemies()
        {
            if (Paused || GameOver) return;

            for (int i = Enemies.Count - 1; i >= 0; i--)
            {
                var epos = Enemies[i];
                int dx = Player.X - epos.X;
                int dy = Player.Y - epos.Y;
                var newPos = epos;
                if (Math.Abs(dy) > Math.Abs(dx)) newPos.Y += Math.Sign(dy);
                else newPos.X += Math.Sign(dx);
                Enemies[i] = newPos;

                if (newPos == Player)
                {
                    GameOver = true;
                    StopTimers();
                    OnGameOver?.Invoke(this, "Egy üldöző elkapott — vesztettél.");
                    return;
                }

                if (Mines.Contains(newPos)) { Enemies.RemoveAt(i); continue; }
            }

            if (Enemies.Count == 0)
            {
                GameOver = true;
                StopTimers();
                OnGameOver?.Invoke(this, "Gratulálok! Nyertél!");
                return;
            }

            OnStateChanged?.Invoke(this, EventArgs.Empty);
        }

        private void CheckCollisions()
        {
            if (Mines.Contains(Player))
            {
                GameOver = true;
                StopTimers();
                OnGameOver?.Invoke(this, "Beléptél egy aknára — vesztettél.");
            }
            else if (Enemies.Any(e => e == Player))
            {
                GameOver = true;
                StopTimers();
                OnGameOver?.Invoke(this, "Egy üldöző elkapott — vesztettél.");
            }//igen tudom,hogy 2x ellenöröz, de van néha késés ha csak egyszer ellenörzöm.
        }

        public void TogglePause()
        {
            Paused = !Paused;
            UpdateTimers();
        }

        private void UpdateTimers()
        {
            if (Paused || GameOver)
                StopTimers();
            else
            {
                enemyTimer.Start();
                clockTimer.Start();
            }
        }

        private void StopTimers()
        {
            enemyTimer.Stop();
            clockTimer.Stop();
        }

        public void SaveGame()
        {
            var data = ConvertToPersistenceData();
            _dataAccess.Save(data);
        }

        public void LoadGame()
        {
            var data = _dataAccess.Load();
            if (data != null)
            {
                LoadFromPersistenceData(data);
            }
        }
    }

namespace Menekulj.Persistence
{
    public class GameStorage : IGameDataAccess
    {
        public event Action? InfoMessage;
        
        private readonly string savePath = Path.Combine(AppContext.BaseDirectory, "savegame.json");

        public void Save(GameData data)
        {
            try
            {
                var options = new JsonSerializerOptions { WriteIndented = true };
                File.WriteAllText(savePath, JsonSerializer.Serialize(data, options));
                InfoMessage?.Invoke("Mentés", "Mentés elkészült: savegame.json");
            }
            catch (Exception ex)
            {
                throw new GameDataException("Mentés sikertelen", ex);
            }
        }

        public GameData? Load()
        {
            try
            {
                if (!File.Exists(savePath))
                {
                    throw new GameDataException("Nincs mentésfájl");
                }

                var json = File.ReadAllText(savePath);
                var data = JsonSerializer.Deserialize(json);
                if (data == null)
                {
                    throw new GameDataException("Hibás mentésfájl formátum");
                }

                InfoMessage?.Invoke("Betöltés", "Betöltés sikeres.");
                return data;
            }
            catch (GameDataException)
            {
                throw;
            }
            catch (Exception ex)
            {
                throw new GameDataException("Betöltés sikertelen", ex);
            }
        }
    }
